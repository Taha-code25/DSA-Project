def return_car():
    global current_user  # Logged-in user

    cars = load_cars_from_csv('cars.csv')
    rentals = load_rentals('rental_history.csv', users, cars)

    active_rental = None
    for rental in rentals:
        if rental.username == current_user.username:
            current_date = datetime.now().date()
            if rental.start_date <= current_date <= rental.end_date or current_date > rental.end_date:
                active_rental = rental
                break

    if not active_rental:
        messagebox.showinfo("Return Car", "You have no car to return.")
        return

    confirm = messagebox.askyesno("Confirm Return", f"Do you want to return {active_rental.car.model}?")
    if not confirm:
        return

    # Late return handling
    current_date = datetime.now().date()
    extra_cost = 0
    if current_date > active_rental.end_date:
        extra_days = (current_date - active_rental.end_date).days
        extra_cost = extra_days * active_rental.car.price_per_day
        current_user.balance -= extra_cost
        messagebox.showwarning("Late Return", f"Late return. Rs.{extra_cost} has been deducted from your balance.")

    # Mark car as available
    for car in cars:
        if car.car_id == active_rental.car.car_id:
            car.is_available = True
            break

    # Clear user's current rental
    current_user.current_rental = None

    # Save updated users and cars
    save_users(users, 'users.csv')
    save_cars_to_csv(cars, 'cars.csv')

    # 🔄 Update rental_history.csv with the new cost
    updated_rows = []
    with open('rental_history.csv', 'r', newline='') as file:
        reader = csv.DictReader(file)
        for row in reader:
            if (row['username'] == current_user.username and
                    row['car_id'] == str(active_rental.car.car_id) and
                    row['start_date'] == str(active_rental.start_date) and
                    row['end_date'] == str(active_rental.end_date)):
                original_cost = float(row['cost'])
                row['cost'] = str(original_cost + extra_cost)
            updated_rows.append(row)

    with open('rental_history.csv', 'w', newline='') as file:
        fieldnames = ['username', 'car_id', 'car_model', 'start_date', 'end_date', 'cost']
        writer = csv.DictWriter(file, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(updated_rows)

    messagebox.showinfo("Return Successful", "Car returned successfully.")

class Person(ABC):
    def __init__(self, username, first_name, last_name):
        self.username = username
        self.first_name = first_name
        self.last_name = last_name

        @abstractmethod
        def get_full_name(self):
                pass

class User(Person):
    def __init__(self, username, password, first_name, last_name, balance):
        self.username = username
        self.password = password
        self.first_name = first_name
        self.last_name = last_name
        self.balance = float(balance)
        self.rental_history = []
        self.current_rental = None

    def get_full_name(self):
        return f"{self.first_name} {self.last_name}"

    def rent_car(self, car, start_date, end_date):
        if self.current_rental:
            print("You already have a car reserved.")
            return False

        total_days = (end_date - start_date).days + 1
        cost = total_days * car.price_per_day

        if self.balance < float(cost):
            print("Insufficient balance.")
            return False

        self.balance -= cost
        self.current_rental = Rental(self.username, car, start_date, end_date, cost)
        self.rental_history.append(self.current_rental)
        save_rental(self.current_rental, 'rental_history.csv')
        car.is_available = False
        print(f"Car {car.model} rented successfully from {start_date} to {end_date}.")
        return True

    def view_rental_history(self):
        if not self.rental_history:
            print("No rental history found.")
        else:
            print("\nRental History:")
            for rental in self.rental_history:
                print(f"Car: {rental.car.model}, From: {rental.start_date}, To: {rental.end_date}, Cost: Rs.{rental.cost}")

    def recharge_balance(self, amount):
        self.balance += amount
        print(f"Balance recharged. New balance: Rs.{self.balance}")

    def return_car(self, current_date):
        if self.current_rental:
            if current_date > self.current_rental.end_date:
                extra_days = (current_date - self.current_rental.end_date).days
                extra_cost = extra_days * self.current_rental.car.price_per_day
                self.balance -= extra_cost
                print(f"Late return. Extra charge of Rs.{extra_cost} deducted.")
            self.current_rental.car.is_available = True
            self.current_rental = None

class Rental:
    def __init__(self, username, car, start_date, end_date, cost):
        self.username = username
        self.car = car
        self.start_date = start_date
        self.end_date = end_date
        self.cost = cost

class Car:
    def __init__(self, car_id, brand, model,seating_capacity, price_per_day, is_available=True):
        self.car_id = int(car_id)
        self.brand = brand
        self.model = model
        self.seating_capacity=seating_capacity
        self.price_per_day = price_per_day
        self.is_available = is_available

    def __str__(self):
        return f"{self.brand} {self.model} - Price/day: Rs.{self.price_per_day}, Available: {'Yes' if self.is_available else 'No'}"

def save_rental(rental, filename):
    file_exists = os.path.exists(filename)

    with open(filename, 'a', newline='') as file:  # newline='' prevents extra/missing newlines
        fieldnames = ['username', 'car_id', 'car_model', 'start_date', 'end_date', 'cost']
        writer = csv.DictWriter(file, fieldnames=fieldnames)

        if not file_exists:
            writer.writeheader()

        writer.writerow({
            'username': rental.username,
            'car_id': rental.car.car_id,
            'car_model': rental.car.model,
            'start_date': rental.start_date,
            'end_date': rental.end_date,
            'cost': rental.cost
        })

def load_rentals(filename, users,cars):
    rentals = []
    if os.path.exists(filename):
        with open(filename, 'r') as file:
            reader = csv.DictReader(file)
            for row in reader:
                username = row['username']
                car_id = int(row['car_id'])
                car = next((c for c in cars if c.car_id == car_id), None)
                if not car:
                    continue  # Skip if car not found

                rental = Rental(
                    username=username,
                    car=car,
                    start_date=datetime.strptime(row['start_date'], "%Y-%m-%d").date(),
                    end_date=datetime.strptime(row['end_date'], "%Y-%m-%d").date(),
                    cost=float(row['cost'])
                )
                rentals.append(rental)

                # Link latest rental to user
                user = next((u for u in users if u.username.strip().lower() == username.strip().lower()), None)
                if user:
                    if user.current_rental is None or rental.start_date > user.current_rental.start_date:
                        user.current_rental = rental
                        user.rental_history.append(rental)
    return rentals

def assign_rentals_to_users(rental_file, users, cars):
    if not os.path.exists(rental_file):
        return

    with open(rental_file, 'r') as file:
        reader = csv.DictReader(file)
        for row in reader:
            username = row['username']
            car_id = int(row['car_id'])

            user = next((u for u in users if u.username.strip().lower() == username.strip().lower()), None)
            car = next((c for c in cars if c.car_id == car_id), None)

            if user and car:
                rental = Rental(
                    username=username,
                    car=car,
                    start_date=datetime.strptime(row['start_date'], "%Y-%m-%d").date(),
                    end_date=datetime.strptime(row['end_date'], "%Y-%m-%d").date(),
                    cost=float(row['cost'])
                )

                user.rental_history.append(rental)

                today = datetime.today().date()
                if rental.start_date <= today <= rental.end_date:
                    user.current_rental = rental

def load_cars_from_csv(filename):
    cars = []
    if os.path.exists(filename):
        with open(filename, 'r') as file:
            reader = csv.DictReader(file)
            for row in reader:
                car = Car(
                    car_id=int(row['car_id']),
                    brand=row['brand'],
                    model=row['model'],
                    seating_capacity=int(row['seating_capacity']),
                    price_per_day=float(row['price_per_day']),  # ✅ Convert here
                    is_available=row['is_available'].lower() == 'true'  # ✅ Convert string to bool
                )
                cars.append(car)
    return cars

def save_cars_to_csv(cars, filename):
    with open(filename, 'w', newline='') as file:
        fieldnames = ['car_id', 'brand', 'model','seating_capacity', 'price_per_day', 'is_available']
        writer = csv.DictWriter(file, fieldnames=fieldnames)
        writer.writeheader()
        for car in cars:
            writer.writerow({field: getattr(car, field) for field in fieldnames})
